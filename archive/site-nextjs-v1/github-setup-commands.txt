# Commands to run after creating GitHub repository
# Replace 'your-repo-name' with the actual repository name you created

# Add the remote origin (replace with your actual repository URL)
git remote add origin https://github.com/Zeyadistired/your-repo-name.git

# Rename the default branch to main (if needed)
git branch -M main

# Push the code to GitHub
git push -u origin main
